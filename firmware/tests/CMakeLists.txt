# Older journal readers against a journal the current build writes (review audit H09).
add_executable(journal_fixture_writer journal_fixture_writer.cpp)
target_link_libraries(journal_fixture_writer PRIVATE monitor_core)
target_compile_options(journal_fixture_writer PRIVATE -Wall -Wextra -Werror -Wpedantic)
foreach(reader tolerant_3c1a01d strict_cefb21b)
    add_executable(journal_reader_${reader} compat/journal_old_reader.cpp compat/journal_${reader}/event_store.cpp ../components/monitor_types/monitor_types.cpp)
    # The older event_store.hpp first; the rest (monitor.hpp) from the current tree.
    target_include_directories(journal_reader_${reader} BEFORE PRIVATE compat/journal_${reader})
    target_include_directories(journal_reader_${reader} PRIVATE ../components/monitor_core/include ../components/sensor_registry/include ../components/alert_service/include ../components/monitor_types/include)
    if(reader STREQUAL "tolerant_3c1a01d")
        target_compile_definitions(journal_reader_${reader} PRIVATE OLDER_READER_TOLERANT=1)
    else()
        target_compile_definitions(journal_reader_${reader} PRIVATE OLDER_READER_TOLERANT=0)
    endif()
    add_test(NAME journal_older_reader_${reader}
             COMMAND sh -c "journal=$($<TARGET_FILE:journal_fixture_writer>) && $<TARGET_FILE:journal_reader_${reader}> \"$journal\"")
endforeach()
